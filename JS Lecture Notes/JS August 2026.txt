const factorial = function fact(n) 
{
	if (n <= 1)
	{
		return 1;
	}	
		
	return n * fact(n - 1);
};

console.log(factorial(5));

function makeCounter() 
{
	let Cnt = 0; // This variable is 'closed over'
	
	return function() 
	{
		Cnt += 100;
		return Cnt;
	};
}

function BankAccount(InitialBalance) 
{
	let Balance = InitialBalance;
	
	return {
		Deposit: (amt1) => { Balance += amt1; return Balance; },
		Withdraw: (amt2) => { Balance -= amt2; return Balance; },
		GetBalance: () => Balance,
	};
}

let User1 = BankAccount(1000);
let User2 = BankAccount(5000);

console.log(User1.GetBalance()); // 1000
console.log(User2.GetBalance()); // 5000

console.log(User1.Deposit(500)); // 1500
console.log(User2.Withdraw(1200)); // 3800

console.log(User1.GetBalance()); // 1500
console.log(User2.GetBalance()); // 3800

function BankAccount(InitialBalance) 
{
	let Balance = InitialBalance;
	
	return {
		Deposit: (amt1) => { Balance += amt1; return `Credited Rs. : ${amt1} | Current Balance : ${Balance}.`; },
		Withdraw: (amt2) => { Balance -= amt2; return `Debited Rs. : ${amt2} | Current Balance : ${Balance}.`; },
		GetBalance: () => `Current Balance : ${Balance}.`,
	};
}

let User1 = BankAccount(1000);
let User2 = BankAccount(5000);

console.log(User1.GetBalance());
console.log(User2.GetBalance());

console.log(User1.Deposit(500));
console.log(User2.Withdraw(1200));

console.log(User1.GetBalance());
console.log(User2.GetBalance());
